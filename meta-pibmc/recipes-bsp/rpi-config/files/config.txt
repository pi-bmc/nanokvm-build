# See https://www.raspberrypi.com/documentation/computers/configuration.html
# Forces the kernel loading system to assume a 64-bit kernel.
arm_64bit=1
# Run as fast as firmware / board allows.
arm_boost=1
# Enable the primary/console UART.
enable_uart=1
# Disable UART for Raspberry Pi 5 to avoid U-boot compatibility issue.

# Automatically load overlays for detected DSI displays
# display_auto_detect=1

# Enable DRM VC4 V3D driver
# dtoverlay=vc4-kms-v3d,noaudio
# max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt.
# Use the kernel's default instead.
# disable_fw_kms_setup=1

# Disable compensation for displays with overscan
# disable_overscan=1

# Disable Bluetooth.
dtoverlay=disable-bt
# Disable Wireless Lan.
dtoverlay=disable-wifi

[pi4]
# Reduce GPU memory to give more to CPU.
gpu_mem=32

# Enable maximum compatibility on both HDMI ports;
# only the one closest to the power/USB-C port will work in practice.
hdmi_safe:0=1
hdmi_safe:1=1
dtoverlay=dwc2,dr_mode=host

[pi5]
usb_max_current_enable=1
enable_uart=0
# Restore CPU thermal monitoring; the Talos kernel's mainline bcm2712 DTB
# omits the AVS monitor and thermal-zones nodes.
dtoverlay=bcm2712-thermal
# UEFI variable store EEPROM on the 40-pin header i2c bus (RP1 i2c1,
# GPIO2/3). U-Boot persists BootOrder/Boot####/BootNext there
# (CONFIG_EFI_VARIABLE_I2C_STORE); the BMC can read/rewrite them out-of-band.
dtoverlay=uefi-eeprom
